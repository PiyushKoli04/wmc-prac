Experiment 02
Simulation of Basic Communication System (Tx - Channel - Rx)
📄 Exp2_A27_WMC.m  —  MATLAB
% MATLAB Code: Basic AM Communication System (Tx - Channel - Rx)
% Student: Prajwal Ashok Zolage | Roll No: A27 | Branch: AI&DS (T.E / A1)
% Aim: Simulation of Basic Communication System using MATLAB

clc;
clear;
close all;

%% Step 1: Generate Message Signal (Transmitter)
Fs = 1000;              % Sampling frequency (Hz)
t = 0:1/Fs:1;          % Time vector (1 second duration)

fm = 5;                % Message frequency = 5 Hz
message = sin(2*pi*fm*t);

%% Step 2: Generate Carrier Signal
fc = 50;               % Carrier frequency = 50 Hz
carrier = sin(2*pi*fc*t);

%% Step 3: AM Modulation
% Standard AM: (1 + m(t)) .* carrier
modulated = (1 + message) .* carrier;

%% Step 4: Communication Channel (Add Noise)
% Add Gaussian random noise to simulate real-world transmission channel
noise = 0.2 * randn(size(modulated));
received = modulated + noise;

%% Step 5: Receiver (Simple Detection / Envelope Detection)
demodulated = abs(received);

%% Step 6: Plot Signals
figure('Name', 'Basic Communication System Simulation');

subplot(5,1,1);
plot(t, message, 'b', 'LineWidth', 1.2);
title('Message Signal');
xlabel('Time (s)');
ylabel('Amplitude');
grid on;

subplot(5,1,2);
plot(t, carrier, 'r', 'LineWidth', 1.2);
title('Carrier Signal');
xlabel('Time (s)');
ylabel('Amplitude');
grid on;

subplot(5,1,3);
plot(t, modulated, 'm', 'LineWidth', 1.2);
title('AM Modulated Signal');
xlabel('Time (s)');
ylabel('Amplitude');
grid on;

subplot(5,1,4);
plot(t, received, 'k', 'LineWidth', 1.1);
title('Received Signal (After Channel Noise)');
xlabel('Time (s)');
ylabel('Amplitude');
grid on;

subplot(5,1,5);
plot(t, demodulated, 'g', 'LineWidth', 1.2);
title('Recovered Signal (Demodulated)');
xlabel('Time (s)');
ylabel('Amplitude');
grid on;
Experiment 03
Demonstration of Frequency Reuse Concept in Cellular Systems
📄 Exp3_A27_WMC.m  —  MATLAB
%% Demonstration of Frequency Reuse Concept in Cellular Systems
% Student: Prajwal Ashok Zolage | Roll No: A27 | Branch: TE/AI&DS (A1)
% Dop: 24/07/2026 | Aim: Simulation of Frequency Reuse in MATLAB

clc;
clear;
close all;

%% 1. Cluster parameters
N = 7;                 % Cluster size (must satisfy N = i^2 + i*j + j^2)
i = 2;
j = 1;                 % Shift parameters for N = 7

%% 2. Generate hexagonal grid (axial coordinates q, r)
range = 4;
size_hex = 1;          % Hexagon "radius" (center to vertex)
colors = lines(N);     % N distinct colors, one per frequency group

figure('Color', 'w', 'Name', 'Frequency Reuse Pattern (N = 7)');
hold on;
axis equal;
axis off;

for q = -range : range
    for r = -range : range
        % Keep a hexagonal region
        if abs(q) <= 4 && abs(r) <= 4 && abs(q + r) <= 4
            
            % Frequency group assignment
            group = mod(q*j - r*i, N);  % 0 to N-1
            
            % Axial -> Cartesian (pointy-top hexagon layout)
            x = size_hex * sqrt(3) * (q + r/2);
            y = size_hex * 1.5 * r;
            
            % Draw hexagon
            drawhexagon(x, y, size_hex, colors(group + 1, :));
            
            % Display frequency group number
            text(x, y, num2str(group + 1), ...
                'HorizontalAlignment', 'center', ...
                'FontWeight', 'bold', ...
                'FontSize', 9);
        end
    end
end

title(['Frequency Reuse Pattern (Cluster Size N = ', num2str(N), ')']);

%% 3. Hexagon drawing function
function drawhexagon(x, y, s, faceColor)
    % Pointy-top orientation
    angles = (0:60:360) + 30;
    xx = x + s * cosd(angles);
    yy = y + s * sind(angles);
    patch(xx, yy, faceColor, ...
        'EdgeColor', 'k', ...
        'LineWidth', 1);
end

Experiment 05
Measurement of Signal Strength vs. Distance for a Wi-Fi Router
📄 Exp5_A27_WMC.m  —  MATLAB
%% Wi-Fi Signal Strength vs Distance Measurement
% Student: Prajwal Ashok Zolage | Roll No: A27 | Branch: TE/AI&DS (A1)
% Dop: 21/08/2026 | Aim: Measurement of Signal Strength vs Distance

clc;
clear;
close all;

% Distance from Wi-Fi source in metres
distance = [0 8 12 17 48];

% Measured RSSI values in dBm
RSSI = [-28 -58 -62 -65 -74];

% Plot graph
figure('Name', 'Wi-Fi RSSI vs Distance');
plot(distance, RSSI, '-o', 'LineWidth', 2, 'MarkerSize', 7);
grid on;

% Labels and title
xlabel('Distance from Wi-Fi Source (m)');
ylabel('Signal Strength (RSSI) (dBm)');
title('Wi-Fi Signal Strength vs Distance');

% Display RSSI values on graph
for i = 1:length(distance)
    text(distance(i), RSSI(i), sprintf('  %.0f dBm', RSSI(i)), ...
        'FontSize', 9, 'FontWeight', 'bold');
end
